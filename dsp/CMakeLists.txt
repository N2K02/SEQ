cmake_minimum_required(VERSION 3.18)
project(seq_reference_dsp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(seq_dsp STATIC src/Biquad.cpp src/Processor.cpp)
target_include_directories(seq_dsp PUBLIC include)
if(MSVC)
  target_compile_options(seq_dsp PRIVATE /W4)
else()
  target_compile_options(seq_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(seq_dsp_tests tests/ProcessorTest.cpp)
target_link_libraries(seq_dsp_tests PRIVATE seq_dsp)
add_test(NAME reference_dsp COMMAND seq_dsp_tests)
